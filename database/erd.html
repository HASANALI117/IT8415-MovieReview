    dbProj_users {
        int user_id PK
        varchar username UK
        varchar email UK
        varchar password_hash "bcrypt/argon hash"
        enum role "viewer | creator | admin"
        tinyint is_active
        timestamp created_at
    }

    dbProj_categories {
        int category_id PK
        varchar name UK
    }

    dbProj_persons {
        int person_id PK
        varchar full_name
        smallint birth_year
        text bio
    }

    dbProj_movies {
        int movie_id PK
        varchar title "FULLTEXT"
        varchar short_description "FULLTEXT"
        text synopsis "FULLTEXT"
        smallint release_year
        int duration_min
        varchar image_url
        varchar media_url
        varchar download_url
        int created_by FK
        tinyint is_published
        timestamp published_at
        int view_count "popularity"
        decimal avg_rating "trigger"
        int rating_count "trigger"
        timestamp created_at
        timestamp updated_at
    }

    dbProj_movie_categories {
        int movie_id PK, FK
        int category_id PK, FK
    }

    dbProj_movie_cast {
        int movie_id PK, FK
        int person_id PK, FK
        enum role PK "director | actor | writer | producer"
        varchar character_name
        int credit_order
    }

    dbProj_ratings {
        int rating_id PK
        int movie_id FK
        int user_id FK
        tinyint stars "1..5"
        timestamp created_at
        timestamp updated_at
    }

    dbProj_comments {
        int comment_id PK
        int movie_id FK
        int user_id FK
        text body
        tinyint is_removed "admin moderation"
        int removed_by FK
        timestamp removed_at
        timestamp created_at
    }

    dbProj_audit_log {
        bigint log_id PK
        varchar table_name
        enum action "INSERT | UPDATE | DELETE"
        int record_id
        varchar detail
        timestamp changed_at
    }

Notes: dbProj_audit_log has no foreign keys — it is an append-only trail written by triggers, so it sits on its own. dbProj_ratings enforces one rating per user per movie via a unique key on (movie_id, user_id). Many-to-many links (movies–categories, movies–persons) are resolved through the two junction tables.
